  var grid = document.getElementById('trialsGrid');

  if (statTotal) statTotal.textContent = trials.length;

  function renderTrials(filtered) {
    grid.innerHTML = '';
    if (filtered.length === 0) {
      grid.style.display = 'none';
      noResults.style.display = 'block';
      resultCount.innerHTML = 'No trials match your current filters.';
      return;
    }
    grid.style.display = '';
    noResults.style.display = 'none';
    resultCount.innerHTML = 'Showing <strong>' + filtered.length + '</strong> of ' + trials.length + ' trials';

    filtered.forEach(function(t) {
      var card = document.createElement('article');
      card.className = 'trial-card';
      var statusClass = t.status === 'recruiting' ? 'recruiting' : (t.status === 'closed' ? 'closed' : 'coming-soon');
      card.innerHTML =
        '<div class="status-bar ' + statusClass + '">' + t.statusLabel + '</div>' +
        '<div class="card-body">' +
          '<h3>' + t.title + '</h3>' +
          '<div class="institution"><span class="ic">●</span> ' + t.institution + '</div>' +
          '<p class="description">' + t.description + '</p>' +
          '<div class="tags">' +
            '<span class="tag-condition">' + t.conditionLabel + '</span>' +
            '<span class="tag-species">' + t.speciesLabel + '</span>' +
            '<span class="tag-location">' + t.state + '</span>' +
          '</div>' +
          '<div class="meta">' +
            '<i>📅</i>' + t.timeline +
            (t.participants ? ' &nbsp;·&nbsp; <i>🐾</i>' + t.participants : '') +
          '</div>' +
        '</div>' +
        '<div class="card-actions">' +
          (t.sourceUrl ? '<a href="' + t.sourceUrl + '" target="_blank" rel="noopener">View source →</a>' : '<a href="../account.html">Register for alerts →</a>') +
          '<span class="source">' + t.source + '</span>' +
        '</div>';
      grid.appendChild(card);
    });
  }

  function applyFilters() {
    var q = searchInput.value.toLowerCase().trim();
    var sp = filterSpecies.value;
    var cond = filterCondition.value;
    var st = filterState.value;
    var stat = filterStatus.value;

    var filtered = trials.filter(function(t) {
      if (sp && t.species !== sp && t.speciesLabel.toLowerCase().indexOf(sp) === -1) return false;
      if (cond && t.condition !== cond) return false;
      if (st && t.state !== st) return false;
      if (stat && t.status !== stat) return false;
      if (q) {
        var searchable = (t.title + ' ' + t.institution + ' ' + t.description + ' ' + t.conditionLabel + ' ' + t.keywords).toLowerCase();
        return searchable.indexOf(q) !== -1;
      }
      return true;
    });

    renderTrials(filtered);
  }

  searchInput.addEventListener('input', applyFilters);
  filterSpecies.addEventListener('change', applyFilters);
  filterCondition.addEventListener('change', applyFilters);
  filterState.addEventListener('change', applyFilters);
  filterStatus.addEventListener('change', applyFilters);

  renderTrials(trials);
})();
